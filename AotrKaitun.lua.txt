nguoitinhcuaeX.LoadData = function()
    local attempts = 0
    local maxAttempts = 10
    while nguoitinhcuaeX.DATA == nil and attempts < maxAttempts do
        SetStatus('Loading functions... (' .. (attempts + 1) .. '/' .. maxAttempts .. ')')
        pcall(function()
            nguoitinhcuaeX.DATA = nguoitinhcuaeX.GET("Functions", "Layout", {})
        end)
        if nguoitinhcuaeX.DATA == nil then
            attempts = attempts + 1
            task.wait(1)
        end
    end
    if nguoitinhcuaeX.DATA == nil then
        SetStatus('Failed to load data, retrying...')
        task.wait(3)
        nguoitinhcuaeX.LoadData()
    end
end
