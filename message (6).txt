local key_file = "VxezeHub/key.txt"

saveKeyToFile = function(key)
    writefile(key_file, key)
end

loadKeyFromFile = function()
    if isfile(key_file) then
        local success, content = pcall(readfile, key_file)
        if success and content and content ~= "" then
            return content
        end
    end
    return nil
end

deleteKeyFile = function()
    if isfile(key_file) then
        delfile(key_file)
    end
end

local API_BASE_URL = "https://vxezehub.click"

local httpRequest = request or http_request or (http and http.request)

local CACHED_HWID = nil
getHWID = function()
    if CACHED_HWID and CACHED_HWID ~= "" then
        return CACHED_HWID
    end
    local ok, id = pcall(function()
        if gethwid then return gethwid() end
        if syn and syn.get_hwid then return syn.get_hwid() end
        return nil
    end)
    if ok and id and tostring(id) ~= "" then
        CACHED_HWID = tostring(id)
        return CACHED_HWID
    end
    local ok2, cid = pcall(function()
        return game:GetService("RbxAnalyticsService"):GetClientId()
    end)
    if ok2 and cid and tostring(cid) ~= "" then
        CACHED_HWID = tostring(cid)
        return CACHED_HWID
    end
    CACHED_HWID = "unknown_" .. tostring(game.Players.LocalPlayer.UserId)
    return CACHED_HWID
end

local function getExecutor()
    local ok, name = pcall(function()
        if identifyexecutor then return (identifyexecutor()) end
        if syn then return "Synapse X" end
        if fluxus then return "Fluxus" end
        if krnl then return "KRNL" end
        if codex then return "Codex" end
        if delta then return "Delta" end
        return nil
    end)
    if ok and name and tostring(name) ~= "" then
        return tostring(name)
    end
    return "unknown"
end

local function trimKey(s)
    if type(s) ~= "string" then return s end
    return (s:gsub("^%s*(.-)%s*$", "%1"))
end

local function safeJSONDecode(body)
    if not body or body == "" then return nil end
    local ok, decoded = pcall(function()
        return HttpService:JSONDecode(body)
    end)
    if ok then return decoded end
    return nil
end

validateKeyWithServer = function(key)
    key = trimKey(key)
    if not key or key == "" then
        return false, "Please enter a key", nil
    end

    local hwid = getHWID()
    local executor = getExecutor()

    local payload = HttpService:JSONEncode({
        key = key,
        hwid = hwid,
        executor = executor
    })

    local result = nil
    local lastError = nil

    for attempt = 1, 3 do
        if httpRequest then
            local ok, response = pcall(function()
                return httpRequest({
                    Url = API_BASE_URL .. "/api/key/check",
                    Method = "POST",
                    Headers = {
                        ["Content-Type"] = "application/json",
                        ["Accept"] = "application/json",
                        ["User-Agent"] = "VxezeHub/2.0 " .. executor
                    },
                    Body = payload
                })
            end)
            if ok and response then
                local code = response.StatusCode or 200
                if code == 429 then
                    return false, "Rate limited, please wait a moment", nil
                end
                if response.Body and code >= 200 and code < 500 then
                    result = safeJSONDecode(response.Body)
                    if result then
                        break
                    end
                end
            end
            if not ok then
                lastError = tostring(response)
            end
        end
        wait(0.6 * attempt)
    end

    if not result then
        for attempt = 1, 2 do
            local ok, body = pcall(function()
                local url = API_BASE_URL .. "/api/validate/key?key=" ..
                    HttpService:UrlEncode(key) .. "&hwid=" .. HttpService:UrlEncode(hwid)
                if httpRequest then
                    local r = httpRequest({
                        Url = url,
                        Method = "GET",
                        Headers = {
                            ["Accept"] = "application/json",
                            ["User-Agent"] = "VxezeHub/2.0 " .. executor
                        }
                    })
                    return r and r.Body
                end
                return game:HttpGet(url)
            end)
            if ok and body then
                result = safeJSONDecode(body)
                if result then
                    break
                end
            end
            wait(0.5 * attempt)
        end
    end

    if result then
        if result.valid == true then
            saveKeyToFile(key)
            return true, result.message or "Valid key", result.time_remaining
        else
            local errorMsg = result.message or "Invalid key"
            if errorMsg:lower():find("hwid") or errorMsg:lower():find("bound") or errorMsg:lower():find("another device") then
                deleteKeyFile()
            end
            return false, errorMsg, result.time_remaining
        end
    end

    return false, "Failed to connect to server. Check your internet." .. (lastError and " (" .. lastError:sub(1, 50) .. ")" or ""), nil
end

loadImage = function(imageId)
    if not imageId or imageId == "" then
        return ""
    end
    if type(imageId) == "string" then
        if imageId:match("^https?://") then
            return imageId
        elseif imageId:match("^rbxassetid://") then
            return imageId
        elseif tonumber(imageId) then
            return "rbxassetid://" .. imageId
        end
    end
    return "rbxassetid://" .. tostring(imageId)
end

local dragging = false
local dragInput = nil
local dragStart = nil
local startPos = nil

smoothDrag = function(input)
    local delta = input.Position - dragStart
    local targetPos = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale,
        startPos.Y.Offset + delta.Y)
    TweenService:Create(MainFrame, TweenInfo.new(0.15, Enum.EasingStyle.Quint, Enum.EasingDirection.Out),
        { Position = targetPos }):Play()
end

MainFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
                dragInput = nil
            end
        end)
    end
end)

MainFrame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        smoothDrag(input)
    end
end)

local KeyVerified = false

autoCheckSavedKey = function()
    local savedKey = loadKeyFromFile()
    if savedKey then
        KeyInput.Text = savedKey
        Notify("Vxeze Hub", "Found saved key, checking with server...", 2)

        local isValid, message, timeInfo = validateKeyWithServer(savedKey)

        if isValid then
            local timeStr = ""
            if timeInfo then
                timeStr = string.format(" (%dh %dm remaining)", timeInfo.hours or 0, timeInfo.minutes or 0)
            end
            Notify("Vxeze Hub", "Key valid!" .. timeStr, 3)

            local greenFlash = Instance.new("Frame")
            greenFlash.BackgroundColor3 = Color3.fromRGB(0, 255, 100)
            greenFlash.BackgroundTransparency = 1
            greenFlash.Size = UDim2.new(1, 0, 1, 0)
            greenFlash.ZIndex = 100
            greenFlash.BorderSizePixel = 0
            greenFlash.Parent = MainFrame

            local greenCorner = Instance.new("UICorner")
            greenCorner.CornerRadius = UDim.new(0, 16)
            greenCorner.Parent = greenFlash

            for i = 1, 2 do
                TweenService:Create(greenFlash, TweenInfo.new(0.08, Enum.EasingStyle.Linear),
                    { BackgroundTransparency = 0.6 }):Play()
                wait(0.08)
                TweenService:Create(greenFlash, TweenInfo.new(0.08, Enum.EasingStyle.Linear),
                    { BackgroundTransparency = 1 }):Play()
                wait(0.08)
            end

            greenFlash:Destroy()
            wait(0.2)

            createSLCloseEffect()
            KeyGui:Destroy()
            KeyVerified = true
            return true
        else
            Notify("Vxeze Hub", "Key invalid: " .. tostring(message), 4)
            deleteKeyFile()
            KeyInput.Text = ""

            TweenService:Create(UIStroke_Input, TweenInfo.new(0.1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                Color = Color3.fromRGB(255, 50, 50)
            }):Play()
            TweenService:Create(KeyInput, TweenInfo.new(0.1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                BackgroundColor3 = Color3.fromRGB(80, 20, 20)
            }):Play()
